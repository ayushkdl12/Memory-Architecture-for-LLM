    users {
        UUID  user_id PK
        TEXT  name
        TIMESTAMPTZ  created_at
    }
    chat_sessions {
        UUID  session_id PK
        UUID  user_id
        TEXT  title
        TIMESTAMPTZ  created_at
        TIMESTAMPTZ  updated_at
    }
    messages {
        UUID  message_id PK
        UUID  session_id
        message_role  role
        TEXT  content
        TIMESTAMPTZ  created_at
    }
    memory_atoms {
        UUID  memory_id PK
        UUID  user_id
        UUID  session_id
        UUID  source_message_id
        memory_type  memory_type
        TEXT  category
        TEXT  subject
        TEXT  attribute
        TEXT  value
        TEXT  content
        priority_level  priority
        DOUBLE  confidence_score
        BOOLEAN  is_confirmed
        BOOLEAN  is_pinned
        BOOLEAN  is_active
        retention_status  retention_status
        INTEGER  access_count
        TIMESTAMPTZ  valid_from
        TIMESTAMPTZ valid_until
        TIMESTAMPTZ expires_at
        TIMESTAMPTZ  created_at
        TIMESTAMPTZ  updated_at
    }
    fact_versions {
        UUID  version_id PK
        UUID  user_id
        TEXT  subject
        TEXT  attribute
        UUID  old_memory_id
        UUID  new_memory_id
        TIMESTAMPTZ  changed_at
        TEXT change_reason
    }
    retrieval_logs {
        UUID  retrieval_id PK
        UUID  session_id
        UUID  message_id
        TEXT  query_text
        JSONB  retrieved_memory_ids
        TEXT retrieval_reason
        TIMESTAMPTZ  created_at
    }
    retention_logs {
        UUID  retention_id PK
        UUID  memory_id
        retention_action  action
        TEXT reason
        DOUBLE  score
        TIMESTAMPTZ  created_at
    }
    media {
        UUID  media_id PK
        UUID  user_id
        UUID  session_id
        UUID  source_message_id
        UUID  memory_id
        TEXT  filename
        TEXT  url
        TEXT  mime_type
        TEXT  description
        TIMESTAMPTZ  created_at
    }
    user_settings {
        UUID  user_id PK
        TEXT  custom_instructions
        TIMESTAMPTZ  updated_at
    }
    search_logs {
        UUID  search_id PK
        UUID  session_id
        UUID  message_id
        TEXT  query_text
        TEXT  provider
        JSONB  results
        TIMESTAMPTZ  created_at
    }
    documents {
        UUID  doc_id PK
        UUID  user_id
        UUID  session_id
        TEXT  filename
        TEXT  mime_type
        BIGINT  char_count
        TEXT  preview
        TIMESTAMPTZ  created_at
    }
    document_chunks {
        UUID  chunk_id PK
        UUID  doc_id
        BIGINT  chunk_index
        TEXT  text
        TIMESTAMPTZ  created_at
    }
